                }
            }

            if (!hit && this.hitWall(this.x, this.y)) { hit = true; }

            if (hit) {
                if (this.shield) this.consumeShield(opponent);
                else this.die();
            }
        }

        consumeShield(opponent) {
            this.shield = false;
            const cx = this.x + this.width / 2;
            const cy = this.y + this.height / 2;
            const clearR = 55;
            this.trail = this.trail.filter(pt => Math.abs(pt.x - cx) >= clearR || Math.abs(pt.y - cy) >= clearR);
            opponent.trail = opponent.trail.filter(pt => Math.abs(pt.x - cx) >= clearR || Math.abs(pt.y - cy) >= clearR);
            this.rebuildGrid();
            opponent.rebuildGrid();
            playSound(crashSound);
        }

        checkPointCollision(point, targetX, targetY) {
            const margin = 5;
            return targetX < point.x + margin && targetX + this.width > point.x - margin &&
                targetY < point.y + margin && targetY + this.height > point.y - margin;
        }

        die(reason) {
            if (!this.alive) return;
            this.alive = false;
            this.deathReason = reason || 'crash';
            explode(this.color, this.x + this.width / 2, this.y + this.height / 2);
            addShake(8);
            vibrate(120);
            playSound(crashSound);
        }

        turn(dx, dy) {
            if (!this.alive || this.isAI) return;
            if (dx === 0 && dy === 0) return;
            if (this.inverted > 0) { dx = -dx; dy = -dy; }
            if (dx === -this.direction.x && dy === -this.direction.y) return;
            if (this.direction.x === 0 && this.direction.y === 0) { this.direction = { x: dx, y: dy }; return; }
            this.direction = { x: dx, y: dy };
        }

        handleKeyDown(key) {
            if (!this.alive || this.isAI) return;
            if (key === this.controls.draft) { this.boosting = false; return; }
            let dx = 0, dy = 0;
            if (key === this.controls.up) dy = -1;
            else if (key === this.controls.down) dy = 1;
            else if (key === this.controls.left) dx = -1;
            else if (key === this.controls.right) dx = 1;
            if (dx !== 0 || dy !== 0) this.turn(dx, dy);
        }

        handleKeyUp(key) {
            if (this.isAI) return;
            if (key === this.controls.draft) this.boosting = true;
        }

        draw() {
            const bw = this.wide > 0 ? this.trailWd + 3 : this.trailWd;

            ctx.strokeStyle = this.color;
            ctx.lineWidth = bw;
            ctx.beginPath();
            if (this.trail.length > 0) {
                ctx.moveTo(this.trail[0].x + 4, this.trail[0].y + 4);
                for (let i = 1; i < this.trail.length; i++) ctx.lineTo(this.trail[i].x + 4, this.trail[i].y + 4);
            }
            ctx.stroke();

            if (this.trail.length > 0) {
                const start = Math.max(0, this.trail.length - 60);
                ctx.save();
                ctx.strokeStyle = this.color;
                ctx.shadowColor = this.color;
                ctx.shadowBlur = 10;
                ctx.lineWidth = bw + 1;
                ctx.globalAlpha = 0.9;
                ctx.beginPath();
                ctx.moveTo(this.trail[start].x + 4, this.trail[start].y + 4);
                for (let i = start + 1; i < this.trail.length; i++) ctx.lineTo(this.trail[i].x + 4, this.trail[i].y + 4);
                ctx.stroke();
                ctx.restore();
            }

            ctx.save();
            ctx.shadowColor = this.color;
            ctx.shadowBlur = 12;
            ctx.fillStyle = this.color;
            ctx.fillRect(this.x, this.y, this.width, this.height);
            ctx.restore();

            const rings = [];
            if (this.shield) rings.push({ color: '#00ff88', r: POWERUP_RADIUS + 4 });
            if (this.boost > 0) rings.push({ color: '#ff9900', r: POWERUP_RADIUS + 2 });
            if (this.inverted > 0) rings.push({ color: '#cc44ff', r: POWERUP_RADIUS + 6 });
            for (const ring of rings) {
                ctx.strokeStyle = ring.color;
                ctx.lineWidth = 2;
                ctx.globalAlpha = 0.7 + 0.3 * Math.sin(Date.now() / 100);
                ctx.beginPath();
                ctx.arc(this.x + this.width / 2, this.y + this.height / 2, ring.r, 0, Math.PI * 2);
                ctx.stroke();
                ctx.globalAlpha = 1;
            }
        }
    }

    // --- MUROS ---
    function generateWalls() {
        walls = [];
        wallPoints = [];
        if (!obstaclesEnabled || !canvas.width) return;
        const margin = 70;
        let attempts = 0;
        while (walls.length < 3 && attempts < 40) {
            attempts++;
            const horizontal = Math.random() < 0.5;
            const len = 90 + Math.random() * 110;
            const x1 = margin + Math.random() * (canvas.width - margin * 2 - len);
            const y1 = margin + Math.random() * (canvas.height - margin * 2 - (horizontal ? 0 : len));
            const x2 = horizontal ? x1 + len : x1;
            const y2 = horizontal ? y1 : y1 + len;
            if (x2 > canvas.width - margin || y2 > canvas.height - margin) continue;
            walls.push({ x1, y1, x2, y2 });
        }
        for (const w of walls) {
            const dist = Math.hypot(w.x2 - w.x1, w.y2 - w.y1);
            const steps = Math.max(1, Math.ceil(dist / 6));
            for (let i = 0; i <= steps; i++) {
                const t = i / steps;
                wallPoints.push({ x: w.x1 + (w.x2 - w.x1) * t, y: w.y1 + (w.y2 - w.y1) * t });
            }
        }
    }

    function drawWalls() {
        for (const w of walls) {
            ctx.save();
            ctx.strokeStyle = '#4d6bff';
            ctx.lineWidth = 10;
            ctx.shadowColor = '#8ab4ff';
            ctx.shadowBlur = 8;
            ctx.beginPath();
            ctx.moveTo(w.x1, w.y1);
            ctx.lineTo(w.x2, w.y2);
            ctx.stroke();
            ctx.restore();
        }
    }

    // --- POWER-UPS ---
    function spawnPowerUp(ttl) {
        const types = Object.keys(POWERUP_DEFS);
        let x = 0, y = 0;
        for (let attempt = 0; attempt < 25; attempt++) {
            const pad = 60;
            x = pad + Math.random() * (canvas.width - pad * 2);
            y = pad + Math.random() * (canvas.height - pad * 2);
            const nearTrail = [player1, player2].some(b => b && b.trail.some(pt => Math.hypot(pt.x - x, pt.y - y) < 45));
            const nearWall = wallPoints.some(pt => Math.hypot(pt.x - x, pt.y - y) < 45);
            if (!nearTrail && !nearWall) break;
        }
        powerUps.push({ type: types[Math.floor(Math.random() * types.length)], x, y, ttl: ttl || POWERUP_TTL, warned: false });
    }

    function ensurePowerUps() {
        while (powerUps.length < POWERUP_COUNT) spawnPowerUp();
    }

    function updatePowerUps() {
        for (let i = powerUps.length - 1; i >= 0; i--) {
            const pu = powerUps[i];
            pu.ttl--;
            if (pu.ttl <= 0) { powerUps.splice(i, 1); continue; }
            if (pu.ttl < 180 && !pu.warned) {
                pu.warned = true;
                playAlarm(880);
            }
            if (pu.ttl === 60) playAlarm(1180);
        }
        ensurePowerUps();
    }

    function drawPowerUps() {
        for (const pu of powerUps) {
            const def = POWERUP_DEFS[pu.type] || POWERUP_DEFS.bonus;
            const nearExpire = pu.ttl < 180;
            const alpha = nearExpire
                ? 0.3 + 0.5 * Math.abs(Math.sin(Date.now() / 70))
                : 0.6 + 0.4 * Math.sin(Date.now() / 150);
            ctx.save();
            ctx.shadowColor = def.color;
            ctx.shadowBlur = 18;
            ctx.globalAlpha = alpha;
            ctx.beginPath();
            ctx.arc(pu.x, pu.y, POWERUP_RADIUS, 0, Math.PI * 2);
            ctx.fillStyle = def.color;
            ctx.fill();
            ctx.globalAlpha = 1;
            ctx.shadowBlur = 0;
            ctx.fillStyle = '#000033';
            ctx.font = 'bold 14px monospace';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText(nearExpire ? '!' : def.label, pu.x, pu.y + 1);
            ctx.restore();
        }
    }

    function checkPowerUps() {
        for (let i = powerUps.length - 1; i >= 0; i--) {
            const pu = powerUps[i];
            for (const p of [player1, player2]) {
                if (!p || !p.alive) continue;
                const reached = pu.x > p.x - POWERUP_RADIUS && pu.x < p.x + p.width + POWERUP_RADIUS &&
                    pu.y > p.y - POWERUP_RADIUS && pu.y < p.y + p.height + POWERUP_RADIUS;
                if (reached) {
                    switch (pu.type) {
                        case 'shield': p.shield = true; break;
                        case 'bonus':
                            if (p === player1) scores.player1++;
                            else scores.player2++;
                            updateScoreDisplay();
                            break;
                        case 'turbo': p.boost = 240; break;
                        case 'wide': p.wide = 360; break;
                        case 'invert': {
                            const other = p === player1 ? player2 : player1;
                            if (!other.isAI) other.inverted = 300;
                            break;
                        }
                    }
                    playSound(powerupSound);
                    vibrate(50);
                    powerUps.splice(i, 1);
                    break;
                }
            }
        }
    }

    // --- EFECTOS ---
    function explode(color, x, y) {
        for (let i = 0; i < 42; i++) {
            const a = Math.random() * Math.PI * 2;
            const spd = 0.8 + Math.random() * 3.4;
            particles.push({ x, y, vx: Math.cos(a) * spd, vy: Math.sin(a) * spd, life: 1, color });
        }
    }

    function updateParticles() {
        for (let i = particles.length - 1; i >= 0; i--) {
            const p = particles[i];
            p.x += p.vx;
            p.y += p.vy;
            p.vx *= 0.96;
            p.vy *= 0.96;
            p.life -= 0.028;
            if (p.life <= 0) { particles.splice(i, 1); continue; }
            ctx.save();
            ctx.shadowColor = p.color;
            ctx.shadowBlur = 8;
            ctx.globalAlpha = Math.max(0, p.life);
            ctx.fillStyle = p.color;
            ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
            ctx.restore();
        }
    }

    function addShake(v) { shakeIntensity = Math.min(16, shakeIntensity + v); }

    function drawGrid() {
        ctx.strokeStyle = 'rgba(0, 255, 255, 0.07)';
        ctx.lineWidth = 1;
        ctx.beginPath();
        for (let x = 20; x < canvas.width; x += 20) { ctx.moveTo(x, 0); ctx.lineTo(x, canvas.height); }
        for (let y = 20; y < canvas.height; y += 20) { ctx.moveTo(0, y); ctx.lineTo(canvas.width, y); }
        ctx.stroke();
    }

    function drawGoalLine() {
        ctx.save();
        ctx.strokeStyle = 'rgba(0,255,255,0.7)';
        ctx.lineWidth = 3;
        ctx.shadowColor = '#00ffff';
        ctx.shadowBlur = 12;
        ctx.beginPath();
        ctx.moveTo(goalX, 0);
        ctx.lineTo(goalX, canvas.height);
        ctx.stroke();
        ctx.fillStyle = '#00ffff';
        ctx.font = 'bold 16px monospace';
        ctx.textAlign = 'center';
        ctx.shadowBlur = 12;
        ctx.fillText('META', goalX, 24);
        ctx.restore();
    }

    // --- GAME ---
    let player1, player2;
    let lastReplay = null;
    let replayActive = false;
    let replayFrame = 0;
    let replayAnimId = null;
    let roundStartTime = 0;
    let roundWarned = false;

    function createBikes() {
        const p1Ctrls = { up: 'w', down: 's', left: 'a', right: 'd', draft: 'shift' };
        const p2Ctrls = { up: 'arrowup', down: 'arrowdown', left: 'arrowleft', right: 'arrowright', draft: 'l' };
        if (gameFormat === 'race') {
            player1 = new Bike(60, canvas.height * 0.38, '#00ffff', p1Ctrls);
            player2 = new Bike(60, canvas.height * 0.62, '#ff00ff', p2Ctrls, gameMode === 'single');
            player1.direction = { x: 1, y: 0 };
            player2.direction = { x: 1, y: 0 };
        } else {
            player1 = new Bike(100, canvas.height / 2, '#00ffff', p1Ctrls);
            player2 = new Bike(canvas.width - 100, canvas.height / 2, '#ff00ff', p2Ctrls, gameMode === 'single');
            player1.direction = { x: 1, y: 0 };
            player2.direction = { x: -1, y: 0 };
        }
        generateWalls();
        freshRound = true;
        particles = [];
        shakeIntensity = 0;
        powerUps = [];
        lastReplay = null;
        roundStartTime = 0;
        roundWarned = false;
    }

    function applyHandicap() {
        if (scores.player1 !== scores.player2) {
            if (scores.player1 < scores.player2 && player1) player1.shield = true;
            else if (scores.player2 < scores.player1 && player2) player2.shield = true;
        }
    }

    function updateScoreDisplay() {
        scoreDisplay.querySelector('.player-1').textContent = `Jugador 1: ${scores.player1}`;
        scoreDisplay.querySelector('.player-2').textContent = `Jugador 2: ${scores.player2}`;
    }

    function updateMatchInfo() {
        const mode = gameMode === 'single' ? 'VS CPU' : '2 JUGADORES';
        const target = matchTarget > 0 ? ` · PRIMERO A ${matchTarget}` : 'MODO INFINITO';
        const fmt = gameFormat === 'race' ? ' · RAZA' : '';
        const bike = ` · MOTO ${BIKE_STYLES[bikeStyle].label}`;
        matchInfoEl.textContent = `MEJOR MARCA: ${record} · ${mode}${target}${fmt}${bike}`;
    }

    function updateEnergyHUD() {
        if (!player1 || !player2) return;
        energy1Bar.style.width = player1.energy + '%';
        energy1Bar.style.background = player1.energy < 25 ? '#ff0033' : (player1.energy < 55 ? '#ffaa00' : '#00ff88');
        energy2Bar.style.width = player2.energy + '%';
        energy2Bar.style.background = player2.energy < 25 ? '#ff0033' : (player2.energy < 55 ? '#ffaa00' : '#ff00ff');
    }

    function gameLoop() {
        if (!gameRunning) return;
        ctx.fillStyle = '#000033';
        ctx.fillRect(-16, -16, canvas.width + 32, canvas.height + 32);

        const sx = shakeIntensity > 0.2 ? (Math.random() * 2 - 1) * shakeIntensity : 0;
        const sy = shakeIntensity > 0.2 ? (Math.random() * 2 - 1) * shakeIntensity : 0;
        ctx.save();
        ctx.translate(sx, sy);

        drawGrid();
        drawWalls();
        if (gameFormat === 'race') drawGoalLine();
        player1.update(player2);
        player2.update(player1);
        if (frameLog) {
            frameLog.push({ x1: player1.x, y1: player1.y, x2: player2.x, y2: player2.y });
            if (frameLog.length > MAX_REPLAY_FRAMES) frameLog.shift();
        }
        updateParticles();
        updatePowerUps();
        drawPowerUps();
        checkPowerUps();
        player1.draw();
        player2.draw();

        ctx.restore();
        shakeIntensity *= 0.88;

        updateEnergyHUD();

        if (roundStartTime) {
            const elapsed = Date.now() - roundStartTime;
            if (elapsed >= ROUND_WARN_AT && !roundWarned) {
                roundWarned = true;
                playAlarm(880);
                playAlarm(700);
                vibrate(250);
            }
            if (roundWarned && elapsed < ROUND_WARN_AT + 3000) {
                ctx.save();
                ctx.globalAlpha = 0.5 + 0.5 * Math.sin(Date.now() / 90);
                ctx.fillStyle = '#ff4444';
                ctx.font = 'bold 24px monospace';
                ctx.textAlign = 'center';
                ctx.fillText('¡CUIDADO!', canvas.width / 2, 30);
                ctx.restore();
            }
        }

        if (gameFormat === 'race') {
            if (player1.alive && player1.x + player1.width >= goalX) { endGame('player1', 'goal'); return; }
            if (player2.alive && player2.x + player2.width >= goalX) { endGame('player2', 'goal'); return; }
        }

        if (!player1.alive || !player2.alive) {
            endGame(null);
            return;
        }
        animationId = requestAnimationFrame(gameLoop);
    }

    function endGame(winnerOverride, reason) {
        gameRunning = false;
        cancelAnimationFrame(animationId);

        let msg = '';
        if (winnerOverride === 'player1') {
            scores.player1++;
            msg = reason === 'goal' ? '¡JUGADOR 1 LLEGA A LA META!' : 'JUGADOR 1 GANA';
            vibrate(200);
        } else if (winnerOverride === 'player2') {
            scores.player2++;
            msg = reason === 'goal' ? '¡JUGADOR 2 LLEGA A LA META!' : 'JUGADOR 2 GANA';
            vibrate(200);
        } else if (!player1.alive && !player2.alive) {
            msg = '¡COLISIÓN SIMULTÁNEA!';
        } else if (!player1.alive) {
            scores.player2++;
            msg = player1.deathReason === 'energy' ? '¡JUGADOR 1 SIN ENERGÍA!' : '¡JUGADOR 1 HA CAÍDO!';
            msg += '<br>GANADOR: JUGADOR 2';
        } else {
            scores.player1++;
            msg = player2.deathReason === 'energy' ? '¡JUGADOR 2 SIN ENERGÍA!' : '¡JUGADOR 2 HA CAÍDO!';
            msg += '<br>GANADOR: JUGADOR 1';
        }

        matchOver = matchTarget > 0 && (scores.player1 >= matchTarget || scores.player2 >= matchTarget);
        if (matchOver) {
            const champ = scores.player1 >= matchTarget ? 'JUGADOR 1' : 'JUGADOR 2';
            msg += `<br><span class="text-warning fs-2">🏆 ${champ} CAMPEÓN</span>`;
            vibrate(300);
        }
        if (!matchOver && (player1.alive || player2.alive)) speedLevel++;

        document.getElementById('gameOverBody').innerHTML = msg;
        updateScoreDisplay();

        saveRecord(matchTarget);
        record = getRecord(matchTarget);
        updateMatchInfo();

        if (frameLog && frameLog.length > 4) {
            lastReplay = frameLog.slice();
            replayBtn.style.display = 'inline-block';
        } else {
            replayBtn.style.display = 'none';
        }

        startBtn.textContent = matchOver ? 'Nueva Partida' : 'Nueva Ronda';
        gameOverModal.show();
    }

    let frameLog = null;

    // --- REPLAY ---
    function startReplay() {
        if (!lastReplay || lastReplay.length < 4) return;
        gameOverModal.hide();
        replayActive = true;
        replayFrame = 0;
        cancelAnimationFrame(replayAnimId);
        replayLoop();
    }
    function replayLoop() {
        if (!replayActive) return;
        if (replayFrame >= lastReplay.length) { replayActive = false; return; }
        ctx.fillStyle = '#000033';
        ctx.fillRect(-16, -16, canvas.width + 32, canvas.height + 32);
        drawGrid();
        drawWalls();
        if (gameFormat === 'race') drawGoalLine();
        drawReplayTrail(0, replayFrame, '#00ffff');
        drawReplayTrail(1, replayFrame, '#ff00ff');
        replayFrame += 3;
        replayAnimId = requestAnimationFrame(replayLoop);
    }
    function drawReplayTrail(idx, upTo, color) {
        ctx.strokeStyle = color;
        ctx.lineWidth = 3;
        ctx.beginPath();
        let started = false;
        for (let i = 0; i <= upTo; i++) {
            const f = lastReplay[i];
            if (!f) break;
            const x = idx === 0 ? f.x1 : f.x2;
            const y = idx === 0 ? f.y1 : f.y2;
            if (!started) { ctx.moveTo(x + 4, y + 4); started = true; }
            else { ctx.lineTo(x + 4, y + 4); }
            if (i >= upTo) break;
        }
        ctx.stroke();
        const lf = lastReplay[upTo];
        if (lf) {
            const hx = idx === 0 ? lf.x1 : lf.x2;
            const hy = idx === 0 ? lf.y1 : lf.y2;
            ctx.fillStyle = color;
            ctx.fillRect(hx, hy, 8, 8);
        }
    }
    function cancelReplay() {
        if (!replayActive) return;
        replayActive = false;
        cancelAnimationFrame(replayAnimId);
        ctx.fillStyle = '#000033';
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        drawGrid();
        drawWalls();
        if (gameFormat === 'race') drawGoalLine();
    }

    // --- CUENTA ATRÁS ---
    function setCountdown(txt, visible) {
        if (visible && txt !== null) countdownText.textContent = txt;
        countdownOverlay.style.display = visible ? 'flex' : 'none';
    }

    function startCountdown() {
        countdownActive = true;
        startBtn.textContent = 'Preparando…';
        playSound(gameStartSound);
        rampMusicFilter(3500);

        let n = 3;
        setCountdown(n, true);
        clearInterval(countdownInterval);
        countdownInterval = setInterval(() => {
            n--;
            if (n <= 0) {
                clearInterval(countdownInterval);
                countdownInterval = null;
                setCountdown('¡YA!', true);
                setTimeout(() => setCountdown(null, false), 450);
                countdownActive = false;
                rampMusicFilter(700);
                gameRunning = true;
                startBtn.textContent = 'Pausar';
                ensurePowerUps();
                frameLog = [];
                roundStartTime = Date.now();
                roundWarned = false;
                gameLoop();
            } else {
                setCountdown(n, true);
            }
        }, 800);
    }

    // --- PAUSA ---
    function pauseGame() {
        if (replayActive) return;
        gameRunning = false;
        if (countdownInterval) {
            clearInterval(countdownInterval);
            countdownInterval = null;
            countdownActive = false;
        }
        setCountdown(null, false);
        cancelAnimationFrame(animationId);
        paused = true;
        pauseMenu.style.display = 'flex';
        startBtn.textContent = 'Continuar';
    }
    function hidePauseMenu() {
        paused = false;
        pauseMenu.style.display = 'none';
    }
    function resumeGame() {
        hidePauseMenu();
        if (!player1 || !player1.alive || !player2.alive) createBikes();
        if (freshRound) {
            freshRound = false;
            startCountdown();
        } else if (player1.alive && player2.alive) {
            gameRunning = true;
            startBtn.textContent = 'Pausar';
            gameLoop();
        } else {
            createBikes();
            freshRound = false;
            startCountdown();
        }
    }

    // --- CONTROLES DE BOTONES ---
    startBtn.addEventListener('click', () => {
        if (replayActive) { cancelReplay(); hidePauseMenu(); return; }
        if (gameRunning || countdownActive) { pauseGame(); return; }
        if (paused) { resumeGame(); return; }
        if (matchOver) {
            matchOver = false;
            speedLevel = 0;
            scores = { player1: 0, player2: 0 };
            updateScoreDisplay();
            createBikes();
            startCountdown();
            return;
        }
        if (!player1 || !player1.alive || !player2.alive) createBikes();
        applyHandicap();
        if (freshRound) {
            freshRound = false;
            startCountdown();
        } else {
            gameRunning = true;
            startBtn.textContent = 'Pausar';
            gameLoop();
        }
    });

        record = getRecord(matchTarget);
        updateMatchInfo();
        ctx.fillStyle = '#000033';
        ctx.fillRect(0, 0, canvas.width, canvas.height);

    function populateRecords() {
        const body = document.getElementById('records-body');
        const fmts = [
            { label: 'INFINITO', value: 0 },
            { label: 'RÁPIDA (3)', value: 3 },
            { label: 'TORNEO (5)', value: 5 }
        ];
        let html = '';
        for (const f of fmts) {
            html += `<tr><td class="text-start small">${f.label}</td>`;
            for (const [mode, ai] of [['single', 'easy'], ['single', 'medium'], ['single', 'hard'], ['multi', 'multi']]) {
                const k = mode + '|' + f.value + (mode === 'single' ? '|' + ai : '|multi');
                let v = 0;
                try { v = Number(localStorage.getItem('tronRecord.' + k)) || 0; } catch (e) { }
                html += `<td class="small">${v}</td>`;
            }
            html += '</tr>';
        }
        body.innerHTML = html;
    }

    document.addEventListener('keydown', (e) => {
        const key = e.key.toLowerCase();
        if (key === 'escape') {
            if (replayActive) { cancelReplay(); return; }
            if (paused || gameRunning || countdownActive) { startBtn.click(); return; }
            return;
        }
        if (['w', 'a', 's', 'd', 'arrowup', 'arrowdown', 'arrowleft', 'arrowright', 'shift', 'l'].includes(key)) e.preventDefault();
        if (player1) player1.handleKeyDown(key);
        if (player2) player2.handleKeyDown(key);
    });

    document.addEventListener('keyup', (e) => {
        const key = e.key.toLowerCase();
        if (player1) player1.handleKeyUp(key);
        if (player2) player2.handleKeyUp(key);
    });

    // Controles táctiles
    let touch = null;
    canvas.addEventListener('touchstart', (e) => {
        if (e.touches.length > 0) {
            const t = e.touches[0];
            touch = { x: t.clientX, y: t.clientY, lx: t.clientX, ly: t.clientY };
        }
        e.preventDefault();
    }, { passive: false });
    canvas.addEventListener('touchmove', (e) => {
        if (touch && e.touches.length > 0) {
            touch.lx = e.touches[0].clientX;
            touch.ly = e.touches[0].clientY;
        }
        e.preventDefault();
    }, { passive: false });
    canvas.addEventListener('touchend', () => {
        if (!touch) return;
        const start = touch;
        touch = null;
        const dx = start.lx - start.x;
        const dy = start.ly - start.y;
        if (Math.abs(dx) < 18 && Math.abs(dy) < 18) return;
        let dir;
        if (Math.abs(dx) > Math.abs(dy)) dir = { x: Math.sign(dx), y: 0 };
        else dir = { x: 0, y: Math.sign(dy) };
        const rect = canvas.getBoundingClientRect();
        const target = gameMode === 'multi'
            ? (start.x < rect.left + canvas.width / 2 ? player1 : player2)
            : player1;
        if (target) target.turn(dir.x, dir.y);
    });
